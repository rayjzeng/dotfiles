#!/bin/bash
set -euo pipefail
trap '' INT

stty -echo -icanon -ixon
printf '\033]0;dotfiles-domain-menu\007\033]1337;SetUserVar=DEVVM_MENU=MQ==\007\033[?25l'
while IFS= read -r frame; do
  [[ $frame == DEVVM_MENU_EXIT ]] && exit 0
  if [[ $frame == 'DEVVM_REQUEST '* ]]; then
    request=${frame#DEVVM_REQUEST }
    encoded=$(printf '%s' "$request" | base64 | tr -d '\n')
    printf '\033]1337;SetUserVar=DEVVM_REQUEST=%s\007' "$encoded"
    continue
  fi
  if [[ $frame == 'DEVVM_PICK '* ]]; then
    encoded=$(printf '%s' "${frame#DEVVM_PICK }" | base64 | tr -d '\n')
    printf '\033]1337;SetUserVar=DEVVM_PICK=%s\007' "$encoded"
    continue
  fi
  if [[ $frame == 'DEVVM_DRAW '* ]]; then
    printf '\033[H\033[2J\033[3J%s' "${frame#DEVVM_DRAW }"
  fi
done
